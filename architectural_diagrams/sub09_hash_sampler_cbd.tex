% Refined bus width tick annotation macros with exact coordinates and zero collision
\newcommand{\buswidthH}[2]{
  \draw[line width=0.6pt, draw=ieeeNavy] ($(#1)+(-2.5pt,-3.5pt)$) -- ($(#1)+(2.5pt,3.5pt)$);
  \node[font=\sffamily\tiny\bfseries, text=ieeeNavy, anchor=south, inner sep=1.5pt] at ($(#1)+(0pt,2.5pt)$) {#2};
}

\newcommand{\buswidthV}[2]{
  \draw[line width=0.6pt, draw=ieeeNavy] ($(#1)+(-3.5pt,-2.5pt)$) -- ($(#1)+(3.5pt,2.5pt)$);
  \node[font=\sffamily\tiny\bfseries, text=ieeeNavy, anchor=east, inner sep=1.5pt] at ($(#1)+(-2pt,0pt)$) {#2};
}
